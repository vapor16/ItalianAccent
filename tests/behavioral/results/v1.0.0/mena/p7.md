        if (s == null) {

    // Versione "a mane", pe ci vole capire comu funziona
    public static String invertiAMane(String s) {
        if (s == null) {
            return null;
        }
        char[] c = s.toCharArray();
        int i = 0, j = c.length - 1;
        while (i < j) {
            char tmp = c[i];
            c[i] = c[j];
            c[j] = tmp;
            i++;
            j--;
        }
        return new String(c);
    }

    public static void main(String[] args) {
        System.out.println(inverti("Lecce"));        // eccel
        System.out.println(invertiAMane("pizzica")); // acizzip
    }
